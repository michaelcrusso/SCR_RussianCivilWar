Trigger("\x004White Army, Siberia/Ukraine"){
Conditions:
	Bring("Current Player", "Fenix (Zealot)", "Anywhere", At least, 1);
	Bring("Current Player", "Protoss High Templar", "Anywhere", At most, 0);
	Deaths("Current Player", "Mature Crysalis", Exactly, 1);
	Bring("\x004White Army, Siberia/Ukraine", "Zerg Cerebrate Daggoth", "Anywhere", Exactly, 0);

Actions:
	Move Location("Current Player", "Fenix (Zealot)", "Anywhere", "General");
	Display Text Message(Always Display, "Volunteers have been deployed to your general's headquarters!");
	Minimap Ping("General");
	Create Unit("Current Player", "Samir Duran (Ghost)", 8, "General");
	Set Deaths("Current Player", "Mature Crysalis", Add, 1);
	Preserve Trigger();
	Comment("General Spawn, 5 mins");
}
