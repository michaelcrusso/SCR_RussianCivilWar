Trigger("All Players"){
Conditions:
	Bring("Current Player", "Men", "Kiev", At least, 6);
	Bring("Current Player", "Buildings", "Kiev", At most, 0);
	Bring("Foes", "Men", "Kiev", At most, 0);
	Bring("Foes", "Buildings", "Kiev", At least, 1);
	Switch("Partisans Kiev", set);

Actions:
	Display Text Message(Always Display, "\x00fWe have captured the city of \x006Kiev!");
	Give Units to Player("Foes", "Current Player", "Buildings", All, "Kiev");
	Minimap Ping("Kiev");
	Set Switch("Partisans Kiev", clear);
	Preserve Trigger();
}
