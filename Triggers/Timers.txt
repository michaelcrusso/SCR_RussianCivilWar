	Always();

	Comment("Set timer at start to 1 second");
}

Trigger("Player 4"){
Conditions:
	Switch("White Movement 2", set);

	Set Deaths("Player 4", "Mineral Chunk (Type 1)", Subtract, 1);
	Preserve Trigger();
	Comment("Timer, 12");
}

Trigger("Player 4"){
Conditions:
	Deaths("Player 4", "Mineral Chunk (Type 1)", At most, 0);
	Switch("White Movement 2", set);

Actions:
	Set Deaths("Player 4", "Mineral Chunk (Type 1)", Add, 12);
	Set Deaths("Player 4", "Mineral Chunk (Type 2)", Add, 1);
	Set Deaths("Player 4", "Mineral Field (Type 1)", Add, 1);
	Preserve Trigger();
	Comment("Type 1 chunk reset second, Type 2 chunk add 1 elapsed second, Type 1 field add general spawn");
}
