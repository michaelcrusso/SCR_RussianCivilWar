Trigger("\x006Soviet Army"){
Conditions:
	Bring("Current Player", "Protoss Zealot", "Anywhere", At least, 1);
	Bring("Current Player", "Protoss High Templar", "Anywhere", At most, 0);
	Deaths("Current Player", "Mature Crysalis", Exactly, 1);
	Bring("\x006Soviet Army", "Terran Armory", "Anywhere", At least, 0);
	Bring("\x006Soviet Army", "Terran Armory", "Anywhere", At most, 4);


Actions:
	Move Location("Current Player", "Protoss Zealot", "Anywhere", "General");
	Display Text Message(Always Display, "Volunteers have been deployed to your general headquarters!");
	Minimap Ping("General");
	Create Unit("Current Player", "Terran Marine", 2, "General");
	Set Deaths("Current Player", "Mature Crysalis", Add, 1);
	Preserve Trigger();
	Comment("Spawn industrial units, move on to rural units");
}
